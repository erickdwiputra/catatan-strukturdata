        return 2

        return 1
    else:
        return -1

def infixToPostfix(s):
    st = []
    res = []

    for c in s:
        # If operand, add to result
        if ('a' <= c <= 'z') or ('A' <= c <= 'Z') or ('0' <= c <= '9'):
            res.append(c)

        # If '(', push to stack
        elif c == '(':
            st.append('(')

        # If ')', pop until '('
        elif c == ')':
            while st and st[-1] != '(':
                res.append(st.pop())
            st.pop()

        # If operator
        else:
            while st and st[-1] != '(' and \
                (prec(st[-1]) > prec(c) or (prec(st[-1]) == prec(c) \
                                    and not isRightAssociative(c))):
                res.append(st.pop())
            st.append(c)

    # Pop remaining operators
    while st:
        res.append(st.pop())

    return ''.join(res)

    print(infixToPostfix(exp))

# function to return precedence of operators
def precedence(c):
    if c == '^':
        return 3
    elif c in ('*', '/'):
        return 2
    elif c in ('+', '-'):
        return 1
    else:
        return -1

# function to check if operator is right-associative
def isRightAssociative(c):
    return c == '^'

# function to check if a character is an operator
def isOperator(c):
    return c in "+-*/^"

# function to convert infix expression to prefix
def infixToPrefix(s):
    st = []
    result = []

    # scan from right to left
    for c in reversed(s):
        if c.isalnum():
            result.append(c)
        elif c == ')':
            st.append(c)
        elif c == '(':
            while st and st[-1] != ')':
                result.append(st.pop())
            if st:
                st.pop()  # remove ')'
        elif isOperator(c):
            while (st and isOperator(st[-1]) and
                  (precedence(st[-1]) > precedence(c) or
                  (precedence(st[-1]) == precedence(c) and isRightAssociative(c)))):
                result.append(st.pop())
            st.append(c)

    # pop remaining operators
    while st:
        result.append(st.pop())

    # reverse at the end to get correct prefix
    return ''.join(reversed(result))

if __name__ == "__main__":
    s = "a*(b+c)/d"
    print(infixToPrefix(s))

        return 2
    else:
        return 0

    if char.isalnum():
        postfix += char

            postfix += stack.pop()

# infix to postfix
def prioritas(op):
    if op in ('+', '-'):
        return 1
    elif op in ('*', '/'):
        return 2
    return 0


def infix_to_postfix(expr):

    stack = []
    output = []

    for ch in expr:

        if ch.isalnum():
            output.append(ch)

        elif ch == '(':
            stack.append(ch)

        elif ch == ')':

            while stack and stack[-1] != '(':
                output.append(stack.pop())

            stack.pop()

        else:

            while stack and prioritas(stack[-1]) >= prioritas(ch):
                output.append(stack.pop())

            stack.append(ch)

    while stack:
        output.append(stack.pop())

    return ''.join(output)


# evaluasi postfix
def evaluasi_postfix(expr):

    stack = []

    for ch in expr:

        if ch.isdigit():
            stack.append(int(ch))

        else:

            b = stack.pop()
            a = stack.pop()

            if ch == '+':
                stack.append(a + b)

            elif ch == '-':
                stack.append(a - b)

            elif ch == '*':
                stack.append(a * b)

            elif ch == '/':
                stack.append(a / b)

    return stack[0]


infix = "2+(3*4)"

postfix = infix_to_postfix(infix)

print("Infix  :", infix)
print("Postfix:", postfix)

hasil = evaluasi_postfix(postfix)

print("Hasil  :", hasil)
